const OnetParser = (function(){
  const LV7 = ['sch','size','loc','prov','dept','reg','nat'];
  const LV11 = ['sch','size','locIn','locOut','prov','area','dept','moeReg','reg','moi','nat'];
  const SUBJ = {'91':'ภาษาไทย','92':'สังคมศึกษา ศาสนา และวัฒนธรรม','93':'ภาษาอังกฤษ','94':'คณิตศาสตร์','95':'วิทยาศาสตร์','01':'ภาษาไทย','02':'สังคมศึกษา ศาสนา และวัฒนธรรม','03':'ภาษาอังกฤษ','04':'คณิตศาสตร์','05':'วิทยาศาสตร์'};
  const CANON={'91':'TH','92':'SO','93':'EN','94':'MA','95':'SC','01':'TH','02':'SO','03':'EN','04':'MA','05':'SC'};
  const PFX = {'91':'ท','92':'ส','93':'ต','94':'ค','95':'ว','01':'ท','02':'ส','03':'ต','04':'ค','05':'ว'};
  const FORM_BY_FILE = [[/IndividualScore/i,1],[/ContentStat/i,5],[/SbjStat/i,6],[/ScoreDistribution/i,4],[/StdQuality/i,3],[/StatbySchool/i,2]];

  function toLines(items){
    const its = items.map(i=>({x:i.x,y:i.y,s:String(i.s||'').trim()})).filter(i=>i.s!=='');
    its.sort((a,b)=>b.y-a.y||a.x-b.x);
    const lines=[];
    its.forEach(it=>{ let l=lines.find(l=>Math.abs(l.y-it.y)<=3); if(!l){ l={y:it.y,items:[]}; lines.push(l); } l.items.push(it); });
    lines.sort((a,b)=>b.y-a.y);
    lines.forEach(l=>{ l.items.sort((a,b)=>a.x-b.x); l.text=l.items.map(i=>i.s).join(' '); });
    return lines;
  }
  // คืน token ตัวเลข (raw) ตามลำดับ; "-" = ไม่มีค่า
  function tokens(str){
    const out=[]; const re=/(^|\s)(-?\d{1,3}(?:,\d{3})+(?:\.\d+)?|-?\d+(?:\.\d+)?|-)(?=\s|$)/g; let m;
    str=String(str).replace(/\*/g,' ');
    while((m=re.exec(str))) out.push(m[2]);
    return out;
  }
  const val = t => t==='-'||t==null ? null : parseFloat(String(t).replace(/,/g,''));
  function labelOf(line){ // ข้อความก่อนตัวเลขตัวแรก
    const parts=[]; for(const it of line.items){ if(/^-?[\d,]+(\.\d+)?$/.test(it.s)) break; parts.push(it.s); } return parts.join(' ').trim();
  }
  function pack(raw){ // 15 token = แบบ ปรนัย/อัตนัย/รวม ; 7 token = แบบรวมอย่างเดียว
    const v=raw.map(val);
    if(v.length===15) return {t:[v[0],v[3],v[6],v[9],v[12],v[13],v[14]], pa:[v[1],v[2]]};
    if(v.length===7)  return {t:v.slice(0,7)};
    return null;
  }
  function meta(lines, fileName){
    const head=lines.slice(0,14); const m={};
    const ft = head.map(l=>l.text).join(' \n ').match(/ฉบ\S*\s*(\d)\s*-/);
    m.form = ft ? +ft[1] : null;
    if(!m.form){ for(const [re,f] of FORM_BY_FILE){ if(re.test(fileName||'')){ m.form=f; break; } } }
    for(const l of head){ const c=l.text.match(/(?:^|\s)(\d{10})(?:\s|$)/); if(c){ m.code=c[1]; const idx=l.items.findIndex(i=>i.s===c[1]); const rest=l.items.slice(idx+2).map(i=>i.s).join(' ').trim(); m.name=rest; break; } }
    for(const l of head){
      if(/ขนาด/.test(l.items[0]&&l.items[0].s) && tokens(l.text).length===0){ m.size=l.items[1]&&l.items[1].s; m.loc=l.items[3]&&l.items[3].s; }
      if(/^\S*จ\S*ง\S*ห\S*ด$/.test((l.items[0]&&l.items[0].s)||'') && tokens(l.text).length===0){ m.prov=l.items[1]&&l.items[1].s; m.dept=l.items.slice(5).map(i=>i.s).join(' '); }
    }
    for(const l of head){ const y=l.text.match(/(\d)\s+\S+\s+(25\d\d)/); if(y){ m.grade=+y[1]; m.year=+y[2]; m.stage=/ธยม|มธ/.test(l.text)?'M':(/ถม/.test(l.text)?'P':null); break; } }
    const fm=(fileName||'').match(/_([MP])(\d)_/i); if(fm){ if(!m.stage) m.stage=fm[1].toUpperCase(); if(!m.grade) m.grade=+fm[2]; }
    const fy=(fileName||'').match(/_(25\d\d)_/); if(fy&&!m.year) m.year=+fy[1];
    if(!m.stage) m.stage='M';
    m.level=m.stage+m.grade;
    // ปรับค่าที่ตั้ง/ขนาดจากข้อความเพี้ยน
    const sz=m.size||''; m.sizeN = /^เล/.test(sz)?'เล็ก':/^กล/.test(sz)?'กลาง':/พ.{0,2}เศ|ศษ/.test(sz)?'ใหญ่พิเศษ':/^ให/.test(sz)?'ใหญ่':sz;
    const lc=m.loc||''; m.locN = /^นอก/.test(lc)?'นอกเมือง':/^ใน/.test(lc)?'ในเมือง':lc;
    return m;
  }
  function parseStatForm(lines, form, warn){
    const subj={}; let cur=null, a=0, expA=(form===6?11:7);
    const keys = form===6?LV11:LV7;
    lines.forEach(l=>{
      const hd=l.text.match(/\((\d{2})\)\s*$/);
      if(hd && l.items[0] && /^\S{1,5}$/.test(l.items[0].s) && l.items.length<=8 && !/\d\.\d/.test(l.text)){ cur=hd[1]; subj[cur]=subj[cur]||{lv:{},std:{},cont:{}}; a=0; return; }
      if(!cur) return;
      const raw=tokens(l.text);
      const sc=l.text.match(/(?:^|\s)([ก-ฮ])\s+(\d+\.\d+)(?=\s)/);
      if(form!==5 && sc && raw.length>=15){ // แถวมาตรฐาน
        const rest=l.text.replace(sc[0],' ');
        const r=tokens(rest).map(val);
        if(r.length===15){ subj[cur].std[sc[1]+sc[2]]=r; }
        return;
      }
      if(!raw.length) return;
      if(a<expA && raw[0] && !/\./.test(raw[0]) && (raw.length===7||raw.length===15)){ const p=pack(raw); if(p){ subj[cur].lv[keys[a]]=p.t; if(p.pa&&a===0) subj[cur].paSch=p.pa; a++; } return; }
      if(form===5 && a>=expA && raw.length===15 && /\./.test(raw[0])){ const lab=labelOf(l); subj[cur].cont[lab]=raw.map(val); return; }
    });
    Object.keys(subj).forEach(c=>{ const n=Object.keys(subj[c].lv).length; if(n!==expA) warn.push('วิชา '+c+': พบแถวระดับ '+n+' จาก '+expA); });
    return subj;
  }
  function parseDist(lines){
    const out={};
    lines.forEach(l=>{
      const f=l.items[0]&&l.items[0].s; if(!/^\d{2}$/.test(f||'')||l.items[0].x>70) return;
      const t=tokens(l.items.slice(1).map(i=>i.s).join(' ')); if(t.length>=12 && t.slice(0,12).every(x=>/^\d+$/.test(x))) out[f]=t.slice(0,12).map(Number);
    });
    return out;
  }
  function parseIndividual(lines){
    const codes=[]; let rows=[];
    lines.forEach(l=>{
      const m=l.text.match(/^\s*(\d+)\s+(\d{8})\s+(\d{13})\s/);
      if(m){ const rest=l.text.slice(m[0].length); rows.push(tokens(rest).map(val)); }
      else { (l.text.match(/(?:^|\s)((?:0|9)\d)(?=\s|$)/g)||[]).forEach(s=>{ const c=s.trim(); if(rows.length===0 && !codes.includes(c)) codes.push(c); }); }
    });
    if(!rows.length||!codes.length) return null;
    const k=codes.length; const T=rows[0].length;
    const sc=T-k-1; const split=(sc===k+2);
    const res={n:rows.length, subj:{}};
    codes.forEach((c,i)=>{
      const ti = split ? (i===0?2:i+2) : i; // ตำแหน่ง "รวม"
      const li = sc + i;
      const sv=rows.map(r=>r[ti]).filter(v=>v!=null); const lv=rows.map(r=>r[li]).filter(v=>v!=null);
      res.subj[c]={n:sv.length, abs:rows.length-sv.length, m:sv.length?+(sv.reduce((a,b)=>a+b,0)/sv.length).toFixed(2):null, lm:lv.length?+(lv.reduce((a,b)=>a+b,0)/lv.length).toFixed(2):null};
    });
    return res;
  }
  function parse(pages, fileName){
    const lines=[]; pages.forEach(p=>toLines(p).forEach(l=>lines.push(l)));
    const warn=[]; const m=meta(pages.length?toLines(pages[0]):[], fileName);
    const out={file:fileName,form:m.form,school:{code:m.code,name:m.name,size:m.sizeN,loc:m.locN,prov:m.prov},year:m.year,level:m.level,stage:m.stage,warn};
    if(!m.code||!m.year||!m.form){ out.error='ไม่ใช่รายงาน O-NET ตามแบบ สทศ. (หา รหัสโรงเรียน/ปีการศึกษา/ฉบับที่ ไม่พบ)'; return out; }
    if(m.form===2||m.form===5||m.form===6) out.subj=parseStatForm(lines,m.form,warn);
    else if(m.form===4) out.dist=parseDist(lines);
    else if(m.form===1){ out.ind=parseIndividual(lines); if(!out.ind) warn.push('อ่านตารางรายบุคคลไม่สำเร็จ'); }
    else if(m.form===3){ out.skipped='ฉบับที่ 3 เป็นข้อความอธิบายระดับคุณภาพ ไม่มีตัวเลขสำหรับวิเคราะห์ (ข้าม)'; }
    return out;
  }
  return {parse, SUBJ, PFX, CANON, LV7, LV11};
})();

function dmerge(a, b) {
  const o = Object.assign({}, a);
  Object.keys(b).forEach(k => {
    const x = o[k], y = b[k];
    if (y && typeof y === 'object' && !Array.isArray(y) && x && typeof x === 'object' && !Array.isArray(x)) o[k] = dmerge(x, y); else o[k] = y;
  });
  return o;
}

}

function supInsights(y, l) {
  const I = idx(), st = getSettings(), b = st.band, prof = {};
  I.schools.forEach(s => prof[s.code] = schoolProfile(y, l, s.code));
  const urgent = I.schools.map(s => {
    const p = prof[s.code]; let pts = 0; const why = [];
    Object.keys(p.subj).forEach(cn => { const d = p.subj[cn].diff; if (d <= -2 * b) { pts += 2; why.push(SUBJ_NAME[cn] + ' ' + sgn(d, 1)); } else if (d <= -b) { pts += 1; why.push(SUBJ_NAME[cn] + ' ' + sgn(d, 1)); } });
    GROUPS.forEach(g => { const a = acadAgg(acadRows(y, { s: s.code, g: g.k })); if (a && a.mean < st.accMid) { pts += 1; why.push(g.k + ' เกรด ' + fmt(a.mean)); } });
    return { s: s.code, pts, why };
  }).filter(x => x.pts > 0).sort((a, b) => b.pts - a.pts);
  const model = I.schools.map(s => ({ s: s.code, p: prof[s.code] })).filter(x => x.p.above >= 2 && x.p.below === 0).sort((a, b) => b.p.totalDiff - a.p.totalDiff);
  const training = SUBJ_ORDER.map(cn => { const a = aggSubject(y, l, cn); return a ? { cn, below: a.rows.filter(r => r.diff != null && r.diff <= -b).length, n: a.rows.length, diff: a.diff } : null; }).filter(Boolean).sort((a, b) => (b.below / b.n) - (a.below / a.n));
  const plc = []; SUBJ_ORDER.forEach(cn => stdAgg(y, l, cn).forEach(s => { if (s.n && s.below / s.n >= .5) plc.push(Object.assign({ cn }, s)); })); plc.sort((a, b) => a.diff - b.diff);
  const follow = []; I.schools.forEach(s => { const why = []; SUBJ_ORDER.forEach(cn => { const tr = trendOf(s.code, l, cn); const f = trendFlags(tr); if (f.some(x => x[0] === 'คะแนนลดลง')) why.push(SUBJ_NAME[cn] + ' คะแนนลดลง'); if (f.some(x => x[0].indexOf('ต่ำกว่าระดับประเทศต่อเนื่อง') === 0)) why.push(SUBJ_NAME[cn] + ' ต่ำกว่าประเทศต่อเนื่อง'); }); if (why.length) follow.push({ s: s.code, why }); });
  follow.sort((a, b) => b.why.length - a.why.length);
  return { urgent, model, training, plc, follow };
}

function csvParse(txt) {
  const rows = []; let r = [], f = '', q = false; txt = txt.replace(/^﻿/, '');
  for (let i = 0; i < txt.length; i++) { const ch = txt[i]; if (q) { if (ch === '"') { if (txt[i + 1] === '"') { f += '"'; i++; } else q = false; } else f += ch; } else if (ch === '"') q = true; else if (ch === ',' || ch === '\t') { r.push(f); f = ''; } else if (ch === '\n') { r.push(f); rows.push(r); r = []; f = ''; } else if (ch !== '\r') f += ch; }
  if (f !== '' || r.length) { r.push(f); rows.push(r); } return rows.filter(x => x.some(c => String(c).trim() !== ''));
}

  downloadBlob('แม่แบบผลการเรียน.csv', 'text/csv;charset=utf-8', '﻿' + rows.map(r => r.map(c => '"' + String(c).replace(/"/g, '""') + '"').join(',')).join('\r\n'));

function matchGroup(s) { s = String(s).replace(/\s+/g, ''); if (/ต่างประเทศ|อังกฤษ/.test(s)) return 'ภาษาต่างประเทศ'; if (/ไทย/.test(s)) return 'ภาษาไทย'; if (/คณิต/.test(s)) return 'คณิตศาสตร์'; if (/วิทย/.test(s)) return 'วิทยาศาสตร์และเทคโนโลยี'; if (/สังคม/.test(s)) return 'สังคมศึกษา ศาสนาและวัฒนธรรม'; if (/สุข|พล/.test(s)) return 'สุขศึกษาและพลศึกษา'; if (/ศิลป/.test(s)) return 'ศิลปะ'; if (/การงาน|อาชีพ/.test(s)) return 'การงานอาชีพ'; return null; }
async function readAcadFile(file) {
  try {
    let rows;
    if (/\.csv$/i.test(file.name)) rows = csvParse(await file.text());
    else { await loadScript('https://cdnjs.cloudflare.com/ajax/libs/xlsx/0.18.5/xlsx.full.min.js', () => !!window.XLSX); const wb = XLSX.read(await file.arrayBuffer(), { type: 'array' }); rows = XLSX.utils.sheet_to_json(wb.Sheets[wb.SheetNames[0]], { header: 1, raw: false, defval: '' }); }
    const hd = rows[0].map(x => String(x).replace(/\s+/g, '')), ix = {}, cc = {};
    hd.forEach((h, i) => { if (/รหัส/.test(h)) ix.code = i; else if (/ปี/.test(h)) ix.year = i; else if (/สาระ/.test(h)) ix.g = i; else if (/จำนวน(นักเรียน|ผู้เรียน)$/.test(h)) ix.n = i; else if (/เฉลี่ย/.test(h)) ix.mean = i; else if (/ผ่าน/.test(h)) ix.pass = i; else if (/(3ขึ้นไป|ระดับ3\+|3\+)/.test(h)) ix.p3 = i; else { const m = h.match(/^(?:ระดับ|ผลการเรียน|เกรด)?(0|1|1\.5|2|2\.5|3|3\.5|4)$/); if (m) cc[m[1]] = i; } });
    const out = [], bad = [];
    rows.slice(1).forEach((r, ri) => {
      const code = String(r[ix.code] || '').trim(), y = parseInt(r[ix.year]), g = matchGroup(r[ix.g]);
      if (!code || !y || !g) { bad.push('แถว ' + (ri + 2) + ': ข้อมูลไม่ครบ (รหัส/ปี/กลุ่มสาระ)'); return; }
      if (!col('schools')[code]) { bad.push('แถว ' + (ri + 2) + ': ไม่พบรหัสโรงเรียน ' + code + ' ในระบบ'); return; }
      const rec = { s: code, y, g };
      if (Object.keys(cc).length >= 6) { rec.c = ACOLS.map(k => cc[k] != null ? (+String(r[cc[k]]).replace(/,/g, '') || 0) : 0); if (!sum(rec.c)) { bad.push('แถว ' + (ri + 2) + ': จำนวนผู้เรียนเป็นศูนย์'); return; } }
      else { rec.n = +String(r[ix.n]).replace(/,/g, '') || 0; rec.mean = num(r[ix.mean]); rec.pass = num(r[ix.pass]); rec.p3 = num(r[ix.p3]); if (rec.mean == null) { bad.push('แถว ' + (ri + 2) + ': ไม่มีคะแนนเฉลี่ย/จำนวนระดับ'); return; } }
      out.push(rec);
    });
    IMP.acadRows = { rows: out, bad }; $('#acPrev').innerHTML = acPreviewHtml();
  } catch (e) { toast(e.message, 'er'); }
}

}

function resizeImage(file, max) {
  return new Promise((res, rej) => {
    const fr = new FileReader(); fr.onerror = rej;
    fr.onload = () => {
      if (/svg/.test(file.type)) return res({ dataUrl: fr.result, blob: file, type: file.type });
      const im = new Image(); im.onerror = () => rej(new Error('อ่านรูปไม่ได้'));
      im.onload = () => { const k = Math.min(1, max / Math.max(im.width, im.height)), c = document.createElement('canvas'); c.width = Math.round(im.width * k); c.height = Math.round(im.height * k); c.getContext('2d').drawImage(im, 0, 0, c.width, c.height); const png = file.type !== 'image/jpeg'; const d = c.toDataURL(png ? 'image/png' : 'image/jpeg', .9); res({ dataUrl: d, type: png ? 'image/png' : 'image/jpeg' }); };
      im.src = fr.result;
    };
    fr.readAsDataURL(file);
  });
}
