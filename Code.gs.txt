function sh_() {

}

  let s = ss.getSheetByName(PIN_SHEET);
  if (!s) { s = ss.insertSheet(PIN_SHEET); s.getRange('A:B').setNumberFormat('@'); s.hideSheet(); }

// อ่านทั้งชีตครั้งเดียว
function readAll_() {
  const s = sh_(), last = s.getLastRow(), out = {};
  if (last < 1) return out;
  s.getRange(1, 1, last, 3).getValues().forEach(r => {
    if (!r[0]) return;
    (out[r[0]] = out[r[0]] || {})[r[1]] = JSON.parse(r[2] || '{}');
  });
  return out;
}

/* ---------- สิทธิ์ผู้ใช้: ผู้ดูแล (ศน.) = ADMIN_PIN, โรงเรียน = PIN ของโรงเรียนนั้น ---------- */
function hash_(pin) {
  const b = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, 'pao-onet|' + String(pin));
  return b.map(x => ('0' + (x & 0xff).toString(16)).slice(-2)).join('');
}
function auth_(authJson) {
  let a = {};
  try { a = typeof authJson === 'string' ? JSON.parse(authJson) : (authJson || {}); } catch (e) { a = {}; }
  const c = CacheService.getScriptCache(), k = 'pinfail';
  const fails = +(c.get(k) || 0);
  if (fails >= 6) throw new Error('PIN_LOCKED: ลองผิดหลายครั้ง กรุณารอ 5 นาที');
  const props = PropertiesService.getScriptProperties();
  const admin = props.getProperty('ADMIN_PIN');
  if (!admin) throw new Error('NO_PIN: ยังไม่ได้ตั้ง PIN ผู้ดูแล — ให้เจ้าของโปรเจกต์รันฟังก์ชัน authorizeOnce ใน Apps Script ก่อน');
  const pin = String(a.pin || '');
  if (pin && pin === admin) return { role: 'admin', sch: '' };
  if (a.sch) {
    const s = pinSh_(), last = s.getLastRow();
    if (last > 0) {
      const rows = s.getRange(1, 1, last, 2).getValues();
      for (let i = 0; i < rows.length; i++) {
        if (String(rows[i][0]) === String(a.sch) && rows[i][1] === hash_(pin)) return { role: 'school', sch: String(a.sch) };
      }
    }
  }
  c.put(k, String(fails + 1), 300);
  throw new Error('PIN_INVALID: PIN ไม่ถูกต้อง');
}
function guard_(a, o) {
  if (a.role === 'admin') return;
  const id = String(o.id || '');
  if (SCHOOL_COLLS.indexOf(o.c) >= 0 && id.split('|')[0] === a.sch) return;
  if (o.c === 'schools' && id === a.sch && o.op === 'merge') {
    const ok = ['name', 'size', 'loc', 'students', 'teachers', 'levels'];
    Object.keys(o.data || {}).forEach(k => { if (ok.indexOf(k) < 0) throw new Error('NOT_ALLOWED: โรงเรียนแก้ไขได้เฉพาะข้อมูลพื้นฐานของตนเอง'); });
    return;
  }
  throw new Error('NOT_ALLOWED: บัญชีโรงเรียนบันทึกได้เฉพาะข้อมูลของโรงเรียนตนเอง');
}

/* ---------- เขียนข้อมูลเป็นชุด (ops) ครั้งเดียว ---------- */
// ops = [{op:'set'|'merge'|'delete', c, id, data}]
function apiCommit(authJson, opsJson) {
  const a = auth_(authJson);
  const ops = JSON.parse(opsJson);
  ops.forEach(o => guard_(a, o));
  const lock = LockService.getScriptLock(); lock.waitLock(25000);
  try {
    const s = sh_(), last = s.getLastRow();
    const v = last > 0 ? s.getRange(1, 1, last, 3).getValues() : [];
    const idx = {}; v.forEach((r, i) => { if (r[0]) idx[r[0] + '|' + r[1]] = i; });
    const dirty = {};
    ops.forEach(o => {
      const k = o.c + '|' + o.id, i = idx[k];
      if (o.op === 'delete') { if (i != null) { v[i] = ['', '', '']; dirty[i] = 1; delete idx[k]; } return; }
      const old = (i != null && o.op === 'merge') ? JSON.parse(v[i][2] || '{}') : {};
      const json = JSON.stringify(deepMerge_(old, o.data || {}));
      if (json.length > 49000) throw new Error('ข้อมูลเกิน 50,000 ตัวอักษร/เซลล์ (' + k + ')');
      if (i != null) { v[i] = [o.c, String(o.id), json]; dirty[i] = 1; }
      else { idx[k] = v.length; dirty[v.length] = 1; v.push([o.c, String(o.id), json]); }
    });
    const di = Object.keys(dirty).map(Number).sort((x, y) => x - y);
    if (di.length > 80 || di.length > v.length * 0.4) {
      const rows = v.filter(r => r[0]);
      s.clearContents();
      if (rows.length) s.getRange(1, 1, rows.length, 3).setValues(rows);
    } else {
      let st = 0;
      while (st < di.length) {
        let en = st; while (en + 1 < di.length && di[en + 1] === di[en] + 1) en++;
        const r0 = di[st], r1 = di[en];
        s.getRange(r0 + 1, 1, r1 - r0 + 1, 3).setValues(v.slice(r0, r1 + 1));
        st = en + 1;
      }
    }
    return JSON.stringify({ ok: true, n: ops.length });
  } finally { lock.releaseLock(); }
}
// merge ระดับ 1 ชั้นลึก: คีย์ที่เป็นออบเจ็กต์ (เช่น subj.lv) รวมเข้าด้วยกัน ที่เหลือเขียนทับ
function deepMerge_(a, b) {
  const out = Object.assign({}, a);
  Object.keys(b).forEach(k => {
    const x = out[k], y = b[k];
    if (y && typeof y === 'object' && !Array.isArray(y) && x && typeof x === 'object' && !Array.isArray(x)) out[k] = deepMerge_(x, y);
    else out[k] = y;
  });
  return out;
}

/* ---------- ตั้ง PIN ---------- */
// payload: {adminPin?, schoolCode?, schoolPin?}
function apiSetPins(authJson, payloadJson) {
  const a = auth_(authJson);
  if (a.role !== 'admin') throw new Error('NOT_ALLOWED: เฉพาะผู้ดูแล');
  const p = JSON.parse(payloadJson);
  if (p.adminPin) {
    if (String(p.adminPin).length < 4) throw new Error('PIN ผู้ดูแลต้องมีอย่างน้อย 4 ตัว');
    PropertiesService.getScriptProperties().setProperty('ADMIN_PIN', String(p.adminPin));
  }
  if (p.schoolCode && p.schoolPin) {
    if (String(p.schoolPin).length < 4) throw new Error('PIN โรงเรียนต้องมีอย่างน้อย 4 ตัว');
    savePin_(p.schoolCode, p.schoolPin);
  }
  return JSON.stringify({ ok: true });
}
// เก็บแฮชใน sheet pins (ใช้ตรวจสิทธิ์) + PIN จริงใน ScriptProperties (ให้ผู้ดูแลดู/ส่ง LINE ได้เท่านั้น)
function savePin_(code, pin) {
  const s = pinSh_(), last = s.getLastRow(), h = hash_(pin);
  let row = 0;
  if (last > 0) s.getRange(1, 1, last, 1).getValues().forEach((r, i) => { if (String(r[0]) === String(code)) row = i + 1; });
  if (!row) row = last + 1;
  s.getRange(row, 1, 1, 2).setValues([[String(code), h]]);
  PropertiesService.getScriptProperties().setProperty('PIN_' + code, String(pin));
}
function genPin_() { return String(100000 + (parseInt(Utilities.getUuid().replace(/-/g, '').slice(0, 8), 16) % 900000)); }
// payload: {codes:[รหัสโรงเรียน], regen:bool} — สุ่ม PIN 6 หลัก (ข้ามโรงเรียนที่มี PIN แล้ว ถ้าไม่สั่ง regen)
function apiGenPins(authJson, payloadJson) {
  const a = auth_(authJson);
  if (a.role !== 'admin') throw new Error('NOT_ALLOWED: เฉพาะผู้ดูแล');
  const p = JSON.parse(payloadJson || '{}'), codes = p.codes || [], have = {}, out = {};
  const s = pinSh_(), last = s.getLastRow();
  if (last > 0) s.getRange(1, 1, last, 1).getValues().forEach(r => { if (r[0]) have[String(r[0])] = 1; });
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    codes.forEach(c => { c = String(c || ''); if (!c || (have[c] && !p.regen)) return; const pin = genPin_(); savePin_(c, pin); out[c] = pin; });
  } finally { lock.releaseLock(); }
  return JSON.stringify({ ok: true, pins: out });
}
function apiGetPins(authJson) {
  const a = auth_(authJson);
  if (a.role !== 'admin') throw new Error('NOT_ALLOWED: เฉพาะผู้ดูแล');
  const props = PropertiesService.getScriptProperties().getProperties(), pins = {};
  Object.keys(props).forEach(k => { if (k.indexOf('PIN_') === 0) pins[k.slice(4)] = props[k]; });
  const s = pinSh_(), last = s.getLastRow();
  const set = last > 0 ? s.getRange(1, 1, last, 1).getValues().map(r => String(r[0])).filter(x => x) : [];
  return JSON.stringify({ ok: true, pins: pins, set: set });
}
// ย้าย PIN เมื่อโรงเรียนเปลี่ยนจากรหัสชั่วคราวเป็นรหัสจริง payload: {from,to}
function apiRekeyPin(authJson, payloadJson) {
  const a = auth_(authJson);
  if (a.role !== 'admin') throw new Error('NOT_ALLOWED: เฉพาะผู้ดูแล');
  const p = JSON.parse(payloadJson || '{}'); if (!p.from || !p.to) return JSON.stringify({ ok: true });
  const s = pinSh_(), last = s.getLastRow(), props = PropertiesService.getScriptProperties();
  if (last > 0) s.getRange(1, 1, last, 1).getValues().forEach((r, i) => { if (String(r[0]) === String(p.from)) s.getRange(i + 1, 1).setValue(String(p.to)); });
  const v = props.getProperty('PIN_' + p.from);
  if (v) { props.setProperty('PIN_' + p.to, v); props.deleteProperty('PIN_' + p.from); }
  return JSON.stringify({ ok: true });
}

/* ---------- อัปโหลดไฟล์ (โลโก้ ฯลฯ) ---------- */
function folder_(name, parent) {
  const it = parent.getFoldersByName(name);
  return it.hasNext() ? it.next() : parent.createFolder(name);
}
// payload: {filename, mimeType, data(base64), section, sub}
// อ่านรูปในโฟลเดอร์ logo เป็น base64 (ใช้วาดปกรายงานบน canvas เมื่อ CORS ของ Drive ไม่ผ่าน)
function apiImgData(id) {
  id = String(id || ''); if (!/^[\w-]{10,}$/.test(id)) throw new Error('รหัสไฟล์ไม่ถูกต้อง');
  const f = DriveApp.getFileById(id), b = f.getBlob(), t = b.getContentType() || '';
  if (!/^image\//.test(t)) throw new Error('ไม่ใช่ไฟล์รูปภาพ');
  const bytes = b.getBytes(); if (bytes.length > 4 * 1024 * 1024) throw new Error('ไฟล์ใหญ่เกินไป');
  let ok = false; const ps = f.getParents(); while (ps.hasNext()) { if (ps.next().getName() === 'logo') ok = true; }
  if (!ok) throw new Error('NOT_ALLOWED');
  return JSON.stringify({ ok: true, type: t, data: Utilities.base64Encode(bytes) });
}

function apiUpload(authJson, json) {
  const a = auth_(authJson);
  if (a.role !== 'admin') throw new Error('NOT_ALLOWED: เฉพาะผู้ดูแลอัปโหลดไฟล์ได้');
  const p = JSON.parse(json);
  let dir = ROOT_FOLDER_ID ? DriveApp.getFolderById(ROOT_FOLDER_ID) : folder_('เว็บแอป-วิเคราะห์ผล-อบจ', DriveApp.getRootFolder());
  if (p.section) dir = folder_(p.section, dir);
  if (p.sub) dir = folder_(p.sub, dir);
  const file = dir.createFile(Utilities.newBlob(Utilities.base64Decode(p.data), p.mimeType, p.filename));
  try { file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (e) {}
  const id = file.getId();
  const url = /^image\//.test(p.mimeType)
    ? 'https://lh3.googleusercontent.com/d/' + id + '=w1600'
    : 'https://drive.google.com/file/d/' + id + '/view';
  return JSON.stringify({ ok: true, id: id, url: url });
}

function updFetch_(path) {
  const r = UrlFetchApp.fetch(UPD_BASE + path + '?t=' + Date.now(), { muteHttpExceptions: true });
  if (r.getResponseCode() !== 200) throw new Error('อ่านศูนย์อัปเดตไม่ได้ (' + r.getResponseCode() + ')');
  return r.getContentText();
}
function updCurrentVersion_() { return APP_VERSION; }
function updLatest_() { return JSON.parse(updFetch_('version.json')); }
function updNewer_(latest, cur) {
  const A = String(latest).split('.').map(Number), B = String(cur).split('.').map(Number);
  for (let i = 0; i < Math.max(A.length, B.length); i++) {
    const x = A[i] || 0, y = B[i] || 0;
    if (x > y) return true; if (x < y) return false;
  }
  return false;
}

    props.setProperty('ADMIN_PIN', pin);
